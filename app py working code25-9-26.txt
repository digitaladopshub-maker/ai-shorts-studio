def parse_time_to_seconds(time_str):
    """ Converts HH:MM:SS, MM:SS, compact digits or plain seconds into total seconds """
    time_str = str(time_str).strip()
    if not time_str:
        return 0
    if ':' in time_str:
        parts = time_str.split(':')
        try:
            parts = [float(p) for p in parts]
        except ValueError:
            return 0
        if len(parts) == 3: # HH:MM:SS
            return parts[0] * 3600 + parts[1] * 60 + parts[2]
        elif len(parts) == 2: # MM:SS
            return parts[0] * 60 + parts[1]
        elif len(parts) == 1:
            return parts[0]
    else:
        clean_digits = ''.join(filter(str.isdigit, time_str))
        if clean_digits.isdigit():
            val = int(clean_digits)
            if len(clean_digits) >= 5: # HHMMSS or HMMSS
                h = val // 10000
                m = (val % 10000) // 100
                s = val % 100
                return h * 3600 + m * 60 + s
            elif len(clean_digits) == 4: # MMSS
                m = val // 100
                s = val % 100
                return m * 60 + s
            else:
                return float(clean_digits)
        try:
            return float(time_str)
        except ValueError:
            return 0
    return 0

def load_audio_to_np(wav_path):
    """ Loads wav audio into numpy array safely using native wave module, bypassing whisper ffmpeg pipe crash """
    try:
        with wave.open(wav_path, 'rb') as wf:
            num_frames = wf.getnframes()
            audio_bytes = wf.readframes(num_frames)
            audio_np = np.frombuffer(audio_bytes, dtype=np.int16).astype(np.float32) / 32768.0
            return audio_np
    except Exception:
        return None

def detect_face_center(v_path, start_sec):
    try:
        cap = cv2.VideoCapture(v_path)
        cap.set(cv2.CAP_PROP_POS_MSEC, int(start_sec) * 1000)
        ret, frame = cap.read()
        cap.release()
        if not ret or frame is None:
            return None
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        cascade_path = cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
        if not os.path.exists(cascade_path):
            return None
        face_cascade = cv2.CascadeClassifier(cascade_path)
        if face_cascade.empty():
            return None
        faces = face_cascade.detectMultiScale(gray, 1.1, 4)
        if len(faces) > 0:
            x, y, w, h = faces[0]
            return x + (w // 2)
    except Exception:
        return None
    return None

            subprocess.run(f'ffmpeg -y -i "{cropped_file}" -vn -acodec pcm_s16le -ar 16000 -ac 1 "{whisper_audio_path}"', shell=True, capture_output=True)
            
            align_map = {"Top (Safe Zone)": "6", "Middle-Center": "5", "Bottom (Safe Zone)": "2"}
            align_val = align_map[caption_align]
            
            text_col, _, _, _, ass_color = get_subtitle_styling(style_preset)
            ass_font_name = get_font_family(font_choice)

            result = None
            if os.path.exists(whisper_audio_path):
                audio_np = load_audio_to_np(whisper_audio_path)
                if audio_np is not None:
                    result = model.transcribe(audio_np, word_timestamps=True)
                else:
                    result = model.transcribe(whisper_audio_path, word_timestamps=True)
            else:
                result = model.transcribe(cropped_file, word_timestamps=True)

            ass_file = os.path.join(DOWNLOAD_DIR, f"subs_{clip_num}.ass")
            
            with open(ass_file, "w", encoding="utf-8") as f:
                f.write(f"[Script Info]\nScriptType: v4.00+\nPlayResX: {scale_w}\nPlayResY: {scale_h}\n\n")
                f.write("[V4+ Styles]\nFormat: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding\n")
                
                margin_v_val = int(scale_h * 0.12) if "Bottom" in caption_align else (int(scale_h * 0.1) if "Top" in caption_align else int(scale_h * 0.5))
                render_ass_fontsize = int(font_size * (scale_h / 480) * 1.5)
                
                margin_lr = int(scale_w * 0.1)
                f.write(f"Style: Default,{ass_font_name},{render_ass_fontsize},&H00FFFFFF,&H000000ff,&H00000000,&H80000000,-1,0,0,0,100,100,0,0,1,4,1,{align_val},{margin_lr},{margin_lr},{margin_v_val},1\n\n")
                f.write("[Events]\nFormat: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text\n")
                
                for segment in result['segments']:
                    if 'words' in segment:
                        words = segment['words']
                        for j in range(0, len(words), words_per_line):
                            chunk = words[j:j + words_per_line]
                            
                            # --- KARAOKE WORD-BY-WORD HIGHLIGHT LOGIC ---
                            if style_preset == "Karaoke Highlight":
                                total_chunk_words = len(chunk)
                                for idx_w, target_word in enumerate(chunk):
                                    w_start = target_word['start']
                                    w_end = target_word['end']
                                    
                                    highlighted_words_list = []
                                    for sub_idx, w_obj in enumerate(chunk):
                                        w_text = w_obj['word'].strip().upper()
                                        if sub_idx == idx_w:
                                            highlighted_words_list.append(r"{\c&H00FFFF&}" + w_text + r"{\c&H00FFFFFF&}")
                                        else:
                                            highlighted_words_list.append(w_text)
                                    
                                    raw_str = " ".join(highlighted_words_list)
                                    render_wrap_chars = max(14, int(30 - (font_size * 0.5)))
                                    wrapped_chunk = textwrap.wrap(raw_str, width=render_wrap_chars)
                                    text_str = "\\N".join(wrapped_chunk)
                                    
                                    s_tot_m, s_s = divmod(w_start, 60)
                                    s_h, s_m = divmod(s_tot_m, 60)
                                    e_tot_m, e_s = divmod(w_end, 60)
                                    e_h, e_m = divmod(e_tot_m, 60)
                                    
                                    s_str = f"{int(s_h)}:{int(s_m):02d}:{int(s_s):02d}.{int((w_start%1)*100):02d}"
                                    e_str = f"{int(e_h)}:{int(e_m):02d}:{int(e_s):02d}.{int((w_end%1)*100):02d}"
                                    
                                    f.write(f"Dialogue: 0,{s_str},{e_str},Default,,0,0,0,,{text_str}\n")
                            else:
                                start_t = chunk[0]['start']
                                end_t = chunk[-1]['end']
                                raw_str = " ".join([w['word'].strip() for w in chunk]).upper()
                                
                                render_wrap_chars = max(14, int(30 - (font_size * 0.5)))
                                wrapped_chunk = textwrap.wrap(raw_str, width=render_wrap_chars)
                                text_str = "\\N".join(wrapped_chunk)
                                
                                s_tot_m, s_s = divmod(start_t, 60)
                                s_h, s_m = divmod(s_tot_m, 60)
                                e_tot_m, e_s = divmod(end_t, 60)
                                e_h, e_m = divmod(e_tot_m, 60)
                                
                                s_str = f"{int(s_h)}:{int(s_m):02d}:{int(s_s):02d}.{int((start_t%1)*100):02d}"
                                e_str = f"{int(e_h)}:{int(e_m):02d}:{int(e_s):02d}.{int((end_t%1)*100):02d}"
                                
                                f.write(f"Dialogue: 0,{s_str},{e_str},Default,,0,0,0,,{text_str}\n")

            # --- ROBUST RELATIVE PATHS FOR FFMPEG ASS FILTER ON WINDOWS ---
            ass_file_rel = os.path.join(DOWNLOAD_DIR, f"subs_{clip_num}.ass").replace('\\', '/')
            fonts_dir_rel = "fonts"
            
            sub_cmd = (
                f'ffmpeg -y -i "{cropped_file}" '
                f'-vf "ass={ass_file_rel}:fontsdir={fonts_dir_rel}" '
                f'-c:v libx264 -preset ultrafast -c:a aac "{final_file}"'
            )
            subprocess.run(sub_cmd, shell=True)
            
            if not os.path.exists(final_file) or os.path.getsize(final_file) == 0:
                if os.path.exists(cropped_file):
                    shutil.copy(cropped_file, final_file)
        else:
            if os.path.exists(final_file):
                os.remove(final_file)
            if os.path.exists(cropped_file):
                os.rename(cropped_file, final_file)

        if os.path.exists(final_file) and os.path.getsize(final_file) > 0:
            st.session_state.generated_clips.append((clip_num, final_file))

        # Update progress bar
        progress_bar.progress((i + 1) / total_tasks)

    status_text.success("🎉 All shorts rendered successfully!")
    st.rerun()
